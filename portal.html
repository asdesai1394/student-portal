document.getElementById('roster-excel-file').addEventListener('change', function(e) { const file = e.target.files[0]; if (!file) return; const reader = new FileReader(); reader.onload = function(event) { try { const data = new Uint8Array(event.target.result); const workbook = XLSX.read(data, { type: 'array' }); const sheetName = workbook.SheetNames[0]; const jsonSheet = XLSX.utils.sheet_to_json(workbook.Sheets[sheetName], { defval: "" }); if (!jsonSheet.length) { alert("The selected Excel file appears to be empty."); return; } temporaryParsedExcelRoster = jsonSheet.map(row => { let rollNo = ""; let studentName = ""; // Iterate through all key-value pairs in the row flexibly Object.keys(row).forEach(key => { const cleanKey = key.trim().toLowerCase().replace(/[^a-z0-javascript]/g, ''); const val = String(row[key]).trim(); // Check for MIS / Roll Number headers if (['mis', 'misno', 'roll', 'rollno', 'prn', 'id', 'srno', 'enrollmentno', 'regno'].some(k => cleanKey.includes(k))) { if (!rollNo && val) rollNo = val; } // Check for Name headers if (['name', 'studentname', 'fullname', 'candidate'].some(k => cleanKey.includes(k))) { if (!studentName && val) studentName = val; } }); // Fallback: If no headers matched, pick 1st column as MIS and 2nd column as Name if (!rollNo || !studentName) { const rowValues = Object.values(row).map(v => String(val).trim()).filter(v => v !== ""); if (rowValues.length >= 2) { if (!rollNo) rollNo = rowValues[0]; if (!studentName) studentName = rowValues[1]; } } return { roll_no: rollNo, student_name: studentName }; }).filter(r => r.roll_no && r.student_name); if (temporaryParsedExcelRoster.length === 0) { alert("Processed 0 student rows.\n\nPlease ensure your Excel sheet has columns for MIS/Roll Number and Student Name."); } else { alert(`✅ Successfully processed ${temporaryParsedExcelRoster.length} student rows!\n\nNow click 'Save / Append Roster Workspace' to finalize.`); } } catch (err) { alert("Error reading Excel file: " + err.message); } }; reader.readAsArrayBuffer(file); });
